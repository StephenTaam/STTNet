156 bool pop(T& out) noexcept(std::is_nothrow_move_assignable_v<T> &&
157 std::is_nothrow_move_constructible_v<T>)
158 {
159 const std::size_t head = head_.load(std::memory_order_relaxed);
160 Slot& slot = buffer_[head & mask_];
161 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
162 const std::intptr_t dif = static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(head + 1);
163
164 if (dif != 0) {
165 // seq != head+1 => empty
166 return false;
167 }
168
169 // Move out
170 out = std::move(*slot.ptr());
171
172 // Destroy in-place
173 slot.destroy();
174
175 // Mark slot as free for producers:
176 // seq = head + capacity
177 slot.seq.store(head + capacity_, std::memory_order_release);
178
179 head_.store(head + 1, std::memory_order_relaxed);
180 return true;
181 }

216 template <class U>
217 bool emplace_impl(U&& v) noexcept(std::is_nothrow_constructible_v<T, U&&>) {
218 std::size_t pos = tail_.load(std::memory_order_relaxed);
219
220 for (;;) {
221 Slot& slot = buffer_[pos & mask_];
222 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
223 const std::intptr_t dif = static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(pos);
224
225 if (dif == 0) {
226 // slot is free for this pos
227 if (tail_.compare_exchange_weak(
228 pos, pos + 1,
229 std::memory_order_relaxed,
230 std::memory_order_relaxed))
231 {
232 // We own this slot now
233 slot.construct(std::forward<U>(v));
234 // Publish to consumer: seq = pos+1 means "ready"
235 slot.seq.store(pos + 1, std::memory_order_release);
236 return true;
237 }
238 // CAS failed: pos updated with current tail; retry
239 } else if (dif < 0) {
240 // slot seq < pos => queue is full (producer wrapped)
241 return false;
242 } else {
243 // Another producer is ahead; move pos forward
244 pos = tail_.load(std::memory_order_relaxed);
245 }
246 }
247 }

249 bool discard_one() noexcept {
250 const std::size_t head = head_.load(std::memory_order_relaxed);
251 Slot& slot = buffer_[head & mask_];
252 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
253 if (static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(head + 1) != 0)
254 return false;
255 slot.destroy();
256 slot.seq.store(head + capacity_, std::memory_order_release);
257 head_.store(head + 1, std::memory_order_relaxed);
258 return true;
259 }
