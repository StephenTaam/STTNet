178class MPSCQueue {
179public:
180 explicit MPSCQueue(std::size_t capacity_pow2)
181 : capacity_(capacity_pow2),
182 mask_(capacity_pow2 - 1),
183 buffer_(capacity_pow2),
184 head_(0),
185 tail_(0)
186 {
187 static_assert(std::is_nothrow_move_constructible_v<T>,
188 "MPSCQueue requires a nothrow-move-constructible value type");
189 // capacity must be power of two
190 if (capacity_ < 2 || (capacity_ & mask_) != 0) {
191 // You can replace with your own assert/log
192 throw std::invalid_argument("MPSCQueue capacity must be power of two and >= 2");
193 }
194
195 // Initialize per-slot sequence
196 for (std::size_t i = 0; i < capacity_; ++i) {
197 buffer_[i].seq.store(i, std::memory_order_relaxed);
198 }
199 }

201 MPSCQueue(const MPSCQueue&) = delete;
202 MPSCQueue& operator=(const MPSCQueue&) = delete;

204 ~MPSCQueue() {
205 // Drain remaining items to call destructors if needed
206 while (discard_one()) {}

214 return emplace_impl(std::move(v));
215 }

217 bool push(const T& v) {
218 // Copy before reserving a ring slot. If the copy throws, producer
219 // progress and the per-slot sequence state remain untouched.
220 T copy(v);
221 return emplace_impl(std::move(copy));
222 }
223
228 bool pop(T& out) noexcept(std::is_nothrow_move_assignable_v<T> &&
229 std::is_nothrow_move_constructible_v<T>)
230 {
231 const std::size_t head = head_.load(std::memory_order_relaxed);
232 Slot& slot = buffer_[head & mask_];
233 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
234 const std::intptr_t dif = static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(head + 1);
235
236 if (dif != 0) {
237 // seq != head+1 => empty
238 return false;
239 }
240
241 // Move out
242 out = std::move(*slot.ptr());
243
244 // Destroy in-place
245 slot.destroy();
246
247 // Mark slot as free for producers:
248 // seq = head + capacity
249 slot.seq.store(head + capacity_, std::memory_order_release);
250
251 head_.store(head + 1, std::memory_order_relaxed);
252 return true;
253 }

259 std::size_t approx_size() const noexcept {
260 const std::size_t t = tail_.load(std::memory_order_acquire);
261 const std::size_t h = head_.load(std::memory_order_acquire);
262 return (t >= h) ? (t - h) : 0;
263 }

269 bool possibly_nonempty() const noexcept {
270 return tail_.load(std::memory_order_acquire) !=
271 head_.load(std::memory_order_acquire);
272 }

275 struct Slot {

277 typename std::aligned_storage<sizeof(T), alignof(T)>::type storage;
278 bool has_value = false;

280 T* ptr() noexcept { return reinterpret_cast<T*>(&storage); }
281 const T* ptr() const noexcept { return reinterpret_cast<const T*>(&storage); }

283 template <class U>
284 void construct(U&& v) noexcept(std::is_nothrow_constructible_v<T, U&&>) {
285 ::new (static_cast<void*>(&storage)) T(std::forward<U>(v));
286 has_value = true;

289 void destroy() noexcept {

297 template <class U>
298 bool emplace_impl(U&& v) noexcept(std::is_nothrow_constructible_v<T, U&&>) {
299 std::size_t pos = tail_.load(std::memory_order_relaxed);
300
301 for (;;) {
302 Slot& slot = buffer_[pos & mask_];
303 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
304 const std::intptr_t dif = static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(pos);
305
306 if (dif == 0) {
307 // slot is free for this pos
308 if (tail_.compare_exchange_weak(
309 pos, pos + 1,
310 std::memory_order_relaxed,
311 std::memory_order_relaxed))
312 {
313 // We own this slot now
314 slot.construct(std::forward<U>(v));
315 // Publish to consumer: seq = pos+1 means "ready"
316 slot.seq.store(pos + 1, std::memory_order_release);
317 return true;
318 }
319 // CAS failed: pos updated with current tail; retry
320 } else if (dif < 0) {
321 // slot seq < pos => queue is full (producer wrapped)
322 return false;
323 } else {
324 // Another producer is ahead; move pos forward
325 pos = tail_.load(std::memory_order_relaxed);
326 }
327 }
328 }
329
330 bool discard_one() noexcept {
331 const std::size_t head = head_.load(std::memory_order_relaxed);
332 Slot& slot = buffer_[head & mask_];
333 const std::size_t seq = slot.seq.load(std::memory_order_acquire);
334 if (static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(head + 1) != 0)
335 return false;
336 slot.destroy();
337 slot.seq.store(head + capacity_, std::memory_order_release);
338 head_.store(head + 1, std::memory_order_relaxed);
339 return true;
340 }

351 alignas(64) std::atomic<std::size_t> tail_;
352};
